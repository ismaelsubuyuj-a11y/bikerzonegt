-- 1. Crear tabla de productos para CONECTATE Guatemala
CREATE TABLE IF NOT EXISTS public.products (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL,
  name TEXT NOT NULL,
  category TEXT NOT NULL,
  price NUMERIC(12, 2) NOT NULL,
  publisher_type TEXT NOT NULL DEFAULT 'official',
  seller_name TEXT NOT NULL,
  seller_phone TEXT,
  condition TEXT NOT NULL DEFAULT 'Nuevo',
  stock INTEGER NOT NULL DEFAULT 1,
  description TEXT,
  image_url TEXT
);

-- 2. Activar seguridad a nivel de fila (RLS)
ALTER TABLE public.products ENABLE ROW LEVEL SECURITY;

-- 3. Políticas públicas para leer y publicar productos
CREATE POLICY "Lectura publica de productos"
ON public.products FOR SELECT USING (true);

CREATE POLICY "Insercion publica de productos"
ON public.products FOR INSERT WITH CHECK (true);
